Skema SQL

    
CREATE TABLE extracted_records (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  nomor_dokumen VARCHAR(50),
  tanggal DATE,
  nama VARCHAR(100),
  instansi VARCHAR(100),
  email VARCHAR(100),
  total BIGINT,
  perlu_review BOOLEAN DEFAULT FALSE,
  dibuat_pada TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
